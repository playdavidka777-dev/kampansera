function createCustomerQR(customerId) {
    const container = document.getElementById("qr-code");

    if (!container) {
        console.error("QR-container saknas.");
        return;
    }

    container.innerHTML = "";

    const url =
        `${window.location.origin}/?customer=${encodeURIComponent(customerId)}`;

    new QRCode(container, {
        text: url,
        width: 256,
        height: 256,
        correctLevel: QRCode.CorrectLevel.H
    });
}
